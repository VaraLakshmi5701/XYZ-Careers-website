CREATE DATBASE job_Portal;
USE job_Portal;

CREATE TABLE jobs (
    id INT NOT NULL AUTO_INCREMENT,
    title VARCHAR(200) NOT NULL,
    location VARCHAR(200),
    salary INT,
    currency VARCHAR(10),
    responsibilities VARCHAR(2000),
    requirements VARCHAR(2000),
    PRIMARY KEY (id)
);
